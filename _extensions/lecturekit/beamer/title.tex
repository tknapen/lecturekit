$-- lecturekit: Beamer title metadata (series/eyebrow shown on the title frame)
$if(title)$
\title$if(short-title)$[$short-title$]$elseif(shorttitle)$[$shorttitle$]$endif${\texorpdfstring{$title$}{$title-meta$}}
$endif$
$if(subtitle)$
\subtitle{\texorpdfstring{$subtitle$}{$subtitle-meta$}}
$endif$
\author$if(shortauthor)$[$shortauthor$]$endif${$for(authors)$$it.name.literal$$sep$ \and $endfor$}
\date$if(shortdate)$[$shortdate$]$endif${$date$}
$if(institute)$
\institute{$for(institute)$$institute$$sep$ \and $endfor$}
$endif$
$if(eyebrow)$
\newcommand{\lkTitleEyebrow}{$eyebrow$}
$elseif(series)$
\newcommand{\lkTitleEyebrow}{$series$}
$endif$
\setbeamertemplate{title page}{%
  \vfill
  \begin{beamercolorbox}[wd=\textwidth,leftskip=0pt]{title}
    \ifdefined\lkTitleEyebrow{\scriptsize\bfseries\color{nmMuted}\addfontfeature{LetterSpace=12}\MakeUppercase{\lkTitleEyebrow}}\par\vskip 0.3cm\fi
    {\usebeamerfont{title}\usebeamercolor[fg]{title}\inserttitle}\par
    \ifx\insertsubtitle\@empty\else\vskip 0.2cm{\usebeamerfont{subtitle}\usebeamercolor[fg]{subtitle}\insertsubtitle}\par\fi
    \vskip 0.9cm
    {\usebeamerfont{author}\usebeamercolor[fg]{author}\insertauthor}\par
    \ifx\insertinstitute\@empty\else\vskip 0.1cm{\usebeamerfont{institute}\usebeamercolor[fg]{institute}\insertinstitute}\par\fi
    \vskip 0.1cm
    {\usebeamerfont{date}\usebeamercolor[fg]{date}\insertdate}\par
  \end{beamercolorbox}
  \vfill\vfill
}
